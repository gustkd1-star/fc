// FCbro 회비 연결 — 이 시트의 "이번 달 납부 현황"만 FCbro 앱으로 보내요 (운영진만 볼 수 있어요).
// 시트 내용은 하나도 바꾸지 않아요. 입금 내역 같은 자세한 기록은 보내지 않아요.
// 사용법: 시트 위 메뉴 "FCbro 앱" → 1. 연결 열쇠 넣기 → 2. 지금 보내기 → 3. 자동 보내기 켜기
var FCB_URL = 'https://dnfocmpsttuaobasisxh.supabase.co/rest/v1/rpc/fee_push';
var FCB_PUBLIC = 'sb_publishable_BSu2zTfGGnMNMDKcDYT6TQ_7q6pMiwZ'; // 공개용 주소 열쇠(비밀 아님)

function onOpen() {
  SpreadsheetApp.getUi().createMenu('FCbro 앱')
    .addItem('1. 연결 열쇠 넣기', 'fcbSetKey')
    .addItem('2. 지금 보내기', 'fcbSend')
    .addItem('3. 자동 보내기 켜기 (1시간마다)', 'fcbAutoOn')
    .addSeparator()
    .addItem('자동 보내기 끄기', 'fcbAutoOff')
    .addToUi();
}

function fcbSetKey() {
  var ui = SpreadsheetApp.getUi();
  var r = ui.prompt('FCbro 연결 열쇠', '앱 → 관리 → 회비에서 만든 열쇠(fcb_로 시작)를 붙여 넣어 주세요.', ui.ButtonSet.OK_CANCEL);
  if (r.getSelectedButton() !== ui.Button.OK) return;
  var k = r.getResponseText().trim();
  if (!/^fcb_[0-9a-f]{64}$/.test(k)) { ui.alert('열쇠 모양이 맞지 않아요. fcb_로 시작하는 긴 글자를 통째로 붙여 넣어 주세요.'); return; }
  PropertiesService.getScriptProperties().setProperty('FCB_KEY', k);
  ui.alert('저장했어요. 이제 메뉴에서 "2. 지금 보내기"를 눌러 보세요.');
}

function fcbSend() {
  var msg = fcbAutoSend();
  try { SpreadsheetApp.getUi().alert(msg); } catch (e) { /* 자동 실행이면 창 없음 */ }
}

function fcbAutoOn() {
  fcbAutoOff();
  ScriptApp.newTrigger('fcbAutoSend').timeBased().everyHours(1).create();
  SpreadsheetApp.getUi().alert('켰어요. 이제 1시간마다 자동으로 보내요. 급하면 "2. 지금 보내기"를 누르면 돼요.');
}

function fcbAutoOff() {
  ScriptApp.getProjectTriggers().forEach(function (t) { if (t.getHandlerFunction() === 'fcbAutoSend') ScriptApp.deleteTrigger(t); });
}

function fcbAutoSend() {
  var key = PropertiesService.getScriptProperties().getProperty('FCB_KEY');
  if (!key) return '먼저 "1. 연결 열쇠 넣기"를 해 주세요.';
  var ss = SpreadsheetApp.getActive(), sheets = ss.getSheets();
  var mem = sheets.filter(function (s) { return /회원관리/.test(s.getName()); })[0];
  var sum = sheets.filter(function (s) { return /요약/.test(s.getName()); })[0];
  if (!mem) return '"회원관리시트" 탭을 찾지 못했어요.';
  var tz = ss.getSpreadsheetTimeZone() || 'Asia/Seoul', now = new Date();
  var y = +Utilities.formatDate(now, tz, 'yyyy'), m = +Utilities.formatDate(now, tz, 'M');
  var data = fcbParse(mem.getDataRange().getDisplayValues(), sum ? sum.getDataRange().getDisplayValues() : [], y, m, mem.getName());
  if (!data.rows.length) return '회원 줄을 하나도 읽지 못했어요. "이름" 칸이 있는 머리 줄을 확인해 주세요.';
  var r = UrlFetchApp.fetch(FCB_URL, { method: 'post', contentType: 'application/json', headers: { apikey: FCB_PUBLIC }, payload: JSON.stringify({ p_key: key, p_data: data }), muteHttpExceptions: true });
  if (r.getResponseCode() >= 300) { var t = r.getContentText(); try { t = JSON.parse(t).message || t; } catch (e) { /* */ } return '보내지 못했어요: ' + t; }
  var owe = data.rows.filter(function (x) { return x.owe > 0; }).length;
  return '보냈어요 · ' + m + '월 ' + data.rows.length + '명 · 미납 ' + owe + '명';
}

// ---- 시트 읽기 (앱 쪽 검사에서도 이 함수를 그대로 써요) ----
function fcbNum(v) { var s = String(v == null ? '' : v).replace(/[,\s원₩명]/g, ''); return /^-?\d+(\.\d+)?$/.test(s) ? Number(s) : 0; }
function fcbParse(memVals, sumVals, year, month, sheetName) {
  var hr = -1;
  for (var i = 0; i < Math.min(memVals.length, 15); i++) { var row = memVals[i].map(function (x) { return String(x).trim(); }); if (row.indexOf('이름') >= 0 && row.indexOf('상태') >= 0) { hr = i; break; } }
  var out = { v: 1, year: year, month: month, at: new Date().toISOString(), sheet: sheetName || '', rows: [], sum: null };
  if (hr < 0) return out;
  var H = memVals[hr].map(function (x) { return String(x).trim(); });
  var col = function (n, from) { for (var j = from || 0; j < H.length; j++) if (H[j] === n) return j; return -1; };
  var iName = col('이름'), iDue = col('납부할금액'), iPaid = col('입금금액'), iOwe = col('납부해야할금액'), iSt = col('상태'), iFine = col('연체벌금');
  var mAmt = col(month + '월'), tStart = Math.max(iSt, col('12월')) + 1, mType = col(month + '월', tStart);
  var tCols = []; for (var q = 1; q <= month; q++) tCols.push(col(q + '월', tStart));
  for (var r = hr + 1; r < memVals.length; r++) {
    var R = memVals[r], name = String(R[iName] || '').trim();
    if (!name || /^(합\s*계|합계|총계)$/.test(name)) continue;
    var oweRaw = iOwe >= 0 ? String(R[iOwe] || '').trim() : '';
    var owe = fcbNum(oweRaw);
    var last = '', lastM = 0; tCols.forEach(function (ci, k) { var t = ci >= 0 ? String(R[ci] || '').trim() : ''; if (t && !/확인/.test(t)) { last = t; lastM = k + 1; } });
    out.rows.push({ name: name, type: mType >= 0 ? String(R[mType] || '').trim() : '', last: last, lastM: lastM, mPaid: mAmt >= 0 ? fcbNum(R[mAmt]) : 0, due: iDue >= 0 ? fcbNum(R[iDue]) : 0, paid: iPaid >= 0 ? fcbNum(R[iPaid]) : 0, owe: owe, oweText: owe ? '' : oweRaw, status: iSt >= 0 ? String(R[iSt] || '').trim() : '', fine: iFine >= 0 ? fcbNum(R[iFine]) : 0 });
  }
  // 요약표의 "당월미입금명단" 상자를 그대로 (총무가 공지하는 그 목록)
  out.list = null;
  for (var la = 0; la < sumVals.length && !out.list; la++) for (var lb = 0; lb < sumVals[la].length; lb++) {
    var cell = String(sumVals[la][lb] || ''), ls = cell.split('\n').map(function (x) { return x.trim(); }).filter(function (x) { return /^.+\s-\s.+$/.test(x); });
    if (ls.length >= 2 && /원/.test(cell)) { out.list = ls.map(function (x) { var p = x.split(/\s-\s/); return { name: p[0].trim(), val: p.slice(1).join(' - ').trim() }; }); break; }
  }
  // 요약표의 이번 달 합계 (참고)
  for (var a = 0; a < Math.min(sumVals.length, 10); a++) {
    var S = sumVals[a].map(function (x) { return String(x).trim(); }), c = S.indexOf(month + '월');
    if (c < 0) continue;
    var pick = function (label) { for (var b = a + 1; b < Math.min(sumVals.length, a + 8); b++) { if (String(sumVals[b][0]).replace(/\s/g, '') === label) return [fcbNum(sumVals[b][c]), fcbNum(sumVals[b][c + 1])]; } return null; };
    out.sum = { fee: pick('월회비'), rest: pick('휴식비') };
    break;
  }
  return out;
}
